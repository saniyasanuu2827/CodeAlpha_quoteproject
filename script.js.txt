/* -------------------------
   Application State
------------------------- */

let currentQuote = null;

let favorites =
    JSON.parse(
        localStorage.getItem("quoteFavorites")
    ) || [];

let history =
    JSON.parse(
        localStorage.getItem("quoteHistory")
    ) || [];

let views =
    Number(
        localStorage.getItem("quoteViews")
    ) || 0;


/* -------------------------
   Local Storage
------------------------- */

function saveData() {

    localStorage.setItem(
        "quoteFavorites",
        JSON.stringify(favorites)
    );

    localStorage.setItem(
        "quoteHistory",
        JSON.stringify(history)
    );

    localStorage.setItem(
        "quoteViews",
        views
    );

}


/* -------------------------
   Random Quote
------------------------- */

function getRandomQuote(list = quotes) {

    if (!list.length) {

        showToast("No matching quotes found.");

        return;

    }

    let randomIndex;

    do {

        randomIndex =
            Math.floor(Math.random() * list.length);

    } while (
        list.length > 1 &&
        list[randomIndex] === currentQuote
    );

    return list[randomIndex];

}


/* -------------------------
   Display Quote
------------------------- */

function displayQuote(quote) {

    if (!quote) return;

    currentQuote = quote;

    quoteText.textContent =
        `"${quote.text}"`;

    quoteAuthor.textContent =
        `— ${quote.author}`;

    category.textContent =
        quote.category;

    updateFavoriteButton();

    quoteCard.style.animation = "none";

    void quoteCard.offsetWidth;

    quoteCard.style.animation =
        "cardIn 0.5s ease";

    views++;

    addToHistory(quote);

    saveData();

    updateStats();

}


/* -------------------------
   New Quote
------------------------- */

function generateNewQuote() {

    const selectedCategory =
        categoryFilter.value;

    let availableQuotes = quotes;

    if (selectedCategory !== "all") {

        availableQuotes =
            quotes.filter(
                quote =>
                    quote.category === selectedCategory
            );

    }

    const quote =
        getRandomQuote(availableQuotes);

    displayQuote(quote);

}


/* -------------------------
   Favorites
------------------------- */

function isFavorite(quote) {

    return favorites.some(
        item =>
            item.text === quote.text &&
            item.author === quote.author
    );

}


function toggleFavorite() {

    if (!currentQuote) return;

    const index =
        favorites.findIndex(
            item =>
                item.text === currentQuote.text &&
                item.author === currentQuote.author
        );

    if (index === -1) {

        favorites.push(currentQuote);

        showToast("Added to favorites ❤️");

    } else {

        favorites.splice(index, 1);

        showToast("Removed from favorites");

    }

    saveData();

    updateFavoriteButton();

    updateStats();

}


function updateFavoriteButton() {

    if (!currentQuote) return;

    if (isFavorite(currentQuote)) {

        favoriteBtn.classList.add("active");

        favoriteBtn.innerHTML =
            '<i class="fa-solid fa-heart"></i>';

    } else {

        favoriteBtn.classList.remove("active");

        favoriteBtn.innerHTML =
            '<i class="fa-regular fa-heart"></i>';

    }

}


/* -------------------------
   History
------------------------- */

function addToHistory(quote) {

    history =
        history.filter(
            item =>
                item.text !== quote.text
        );

    history.unshift(quote);

    history =
        history.slice(0, 10);

}


function renderHistory() {

    if (!history.length) {

        historyList.innerHTML = `
            <div class="empty-state">
                <i class="fa-solid fa-clock"></i>
                <p>No quote history yet.</p>
            </div>
        `;

        return;

    }

    historyList.innerHTML =
        history.map(quote => `

            <div class="modal-quote">

                <p>"${quote.text}"</p>

                <small>
                    — ${quote.author}
                </small>

            </div>

        `).join("");

}


/* -------------------------
   Favorites List
------------------------- */

function renderFavorites() {

    if (!favorites.length) {

        favoritesList.innerHTML = `
            <div class="empty-state">

                <i class="fa-regular fa-heart"></i>

                <p>
                    You haven't saved any quotes yet.
                </p>

            </div>
        `;

        return;

    }

    favoritesList.innerHTML =
        favorites.map(quote => `

            <div class="modal-quote">

                <p>"${quote.text}"</p>

                <small>
                    — ${quote.author}
                </small>

            </div>

        `).join("");

}


/* -------------------------
   Copy Quote
------------------------- */

async function copyQuote() {

    if (!currentQuote) return;

    const text =
        `"${currentQuote.text}" — ${currentQuote.author}`;

    try {

        await navigator.clipboard.writeText(text);

        showToast("Quote copied to clipboard!");

    } catch {

        showToast("Unable to copy quote.");

    }

}


/* -------------------------
   Share Quote
------------------------- */

async function shareQuote() {

    if (!currentQuote) return;

    const text =
        `"${currentQuote.text}" — ${currentQuote.author}`;

    if (navigator.share) {

        try {

            await navigator.share({

                title: "QuoteFlow",

                text: text

            });

        } catch {

            // User cancelled sharing

        }

    } else {

        await copyQuote();

        showToast(
            "Sharing unavailable. Quote copied!"
        );

    }

}


/* -------------------------
   Search
------------------------- */

function searchQuotes() {

    const search =
        searchInput.value
            .toLowerCase()
            .trim();

    const selectedCategory =
        categoryFilter.value;

    let results =
        quotes.filter(quote => {

            const matchesSearch =
                quote.text
                    .toLowerCase()
                    .includes(search) ||

                quote.author
                    .toLowerCase()
                    .includes(search);

            const matchesCategory =
                selectedCategory === "all" ||
                quote.category === selectedCategory;

            return matchesSearch &&
                   matchesCategory;

        });

    if (results.length) {

        displayQuote(
            getRandomQuote(results)
        );

    } else {

        showToast("No quotes found.");

    }

}


/* -------------------------
   Theme
------------------------- */

function toggleTheme() {

    document.body.classList.toggle("dark");

    const isDark =
        document.body.classList.contains("dark");

    localStorage.setItem(
        "quoteTheme",
        isDark ? "dark" : "light"
    );

    themeBtn.innerHTML =
        isDark

        ? '<i class="fa-solid fa-sun"></i>'

        : '<i class="fa-solid fa-moon"></i>';

}


function loadTheme() {

    const theme =
        localStorage.getItem("quoteTheme");

    if (theme === "dark") {

        document.body.classList.add("dark");

        themeBtn.innerHTML =
            '<i class="fa-solid fa-sun"></i>';

    }

}


/* -------------------------
   Toast
------------------------- */

let toastTimer;

function showToast(message) {

    toastMessage.textContent =
        message;

    toast.classList.add("show");

    clearTimeout(toastTimer);

    toastTimer =
        setTimeout(() => {

            toast.classList.remove("show");

        }, 2500);

}


/* -------------------------
   Statistics
------------------------- */

function updateStats() {

    totalQuotes.textContent =
        quotes.length;

    favoriteCount.textContent =
        favorites.length;

    totalFavorites.textContent =
        favorites.length;

    viewCount.textContent =
        views;

}

    "keydown",
    event => {

        if (event.key === "Enter") {

            searchQuotes();

        }

    }
);


categoryFilter.addEventListener(
    "change",
    generateNewQuote
);


/* -------------------------
   Keyboard Shortcut
------------------------- */

document.addEventListener(
    "keydown",
    event => {

        // Don't generate quote while typing
        if (
            event.target.tagName === "INPUT" ||
            event.target.tagName === "TEXTAREA"
        ) {
            return;
        }

        if (event.code === "Space") {

            event.preventDefault();

            generateNewQuote();

        }

        if (event.key === "Escape") {

            favoritesModal.classList.remove("show");

            historyModal.classList.remove("show");

        }

    }
);


/* -------------------------
   Application Start
------------------------- */

function init() {

    loadTheme();

    updateStats();

    generateNewQuote();

}


init();
